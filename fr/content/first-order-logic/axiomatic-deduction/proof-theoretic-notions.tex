\olsection{Notions de théorie de la démonstration}

\begin{explain}
De même que nous avons défini plusieurs notions sémantiques importantes
(\iftag{FOL}{validité}{tautologie}, conséquence logique et
satisfaisabilité), nous définissons maintenant les \emph{notions de
théorie de la démonstration} correspondantes. Celles-ci ne font pas
appel à la satisfaction d'!!{sentence}s dans des !!{structure}s, mais
à la !!{derivability} ou à la non-!!{derivability} de certaines
!!{formula}s. La coïncidence de ces notions constitue une découverte
importante : c'est le contenu des \emph{théorèmes de correction et de
complétude}.
\end{explain}

\begin{defn}[!!^{derivability}]
!!^a{formula}~$!A$ est \emph{!!{derivable} à partir de~$\Gamma$}, ce
que l'on note $\Gamma \Proves !A$, s'il existe une !!{derivation} à
partir de~$\Gamma$ qui se termine par~$!A$.
\end{defn}

\begin{defn}[Théorèmes]
!!^a{formula}~$!A$ est un \emph{théorème} s'il existe une
!!{derivation} de~$!A$ à partir de l'ensemble vide. On écrit
$\Proves !A$ si $!A$ est un théorème, et $\Proves/ !A$ dans le cas
contraire.
\end{defn}

\begin{defn}[Cohérence]
Un ensemble~$\Gamma$ de !!{formula}s est \emph{cohérent} si et seulement
si $\Gamma \Proves/ \lfalse$; il est \emph{incohérent} dans le cas
contraire.
\end{defn}

\begin{prop}[Réflexivité]
\ollabel{prop:reflexivity}
Si $!A \in \Gamma$, alors $\Gamma \Proves !A$.
\end{prop}

\begin{proof}
La !!{formula}~$!A$ constitue à elle seule une !!{derivation} de~$!A$
à partir de~$\Gamma$.
\end{proof}

\begin{prop}[Monotonie]
\ollabel{prop:monotonicity}
Si $\Gamma \subseteq \Delta$ et $\Gamma \Proves !A$, alors
$\Delta \Proves !A$.
\end{prop}

\begin{proof}
Une !!{derivation} à partir de~$\Gamma$ reste une !!{derivation} à
partir de~$\Delta$, puisque toutes les hypothèses qu'elle utilise
appartiennent à~$\Delta$ et que les justifications des lignes sont
inchangées.
\end{proof}

\iftag{FOL}{
\begin{lem}[Renommage des constantes propres]
\ollabel{lem:renaming-parameters}
Soit $S$ un ensemble fini de constantes. Toute !!{derivation} peut
être remplacée par une !!{derivation} de même conclusion, dont les
hypothèses sont des hypothèses de la !!{derivation} initiale et dont
les constantes propres sont absentes de~$S$.
\end{lem}

\begin{proof}
Ajoutons d'abord à~$S$ les constantes des hypothèses initiales.
Procédons par récurrence sur la longueur. Si la dernière ligne est
une hypothèse ou un axiome, cette seule ligne suffit. Si elle résulte
d'un modus ponens, appliquons l'hypothèse de récurrence aux deux
préfixes qui se terminent par les prémisses. Réunissons les deux
!!{derivation}s obtenues et appliquons le modus ponens. Comme leurs
constantes propres évitent~$S$, les hypothèses réunies respectent
encore la condition de fraîcheur.

Si la dernière ligne résulte de~\QR, soit $a$ sa constante propre.
Si $a$ appartient à~$S$, échangeons uniformément $a$ avec une constante
$b$ absente de la !!{derivation} et de~$S$. La règle garantit que
l'échange fixe les hypothèses et la conclusion de l'inférence.
Appliquons l'hypothèse de récurrence au préfixe qui établit la
prémisse, puis appliquons~\QR{} avec la constante ainsi choisie.
\end{proof}
}{}

\begin{prop}[Transitivité]
\ollabel{prop:transitivity}
Si $\Gamma \Proves !A$ et $\{!A\} \cup \Delta \Proves !B$, alors
$\Gamma \cup \Delta \Proves !B$.
\end{prop}

\begin{proof}
Soit $!B_1$, \dots, $!B_l = !B$ une !!{derivation} à partir de
$\{!A\} \cup \Delta$. Soit $!A_1$, \dots, $!A_k = !A$ une
!!{derivation} à partir de~$\Gamma$.
\iftag{FOL}{Par le \olref{lem:renaming-parameters}, choisissons les
constantes propres des deux !!{derivation}s en dehors de l'ensemble
fini des constantes de leurs hypothèses réunies.}{}
Considérons la suite
\[
!A_1, \dots, !A_k = !A, !B_1, \dots, !B_l = !B.
\]
Renouvelons à chaque ligne $!B_i$ qui prenait $!A$ pour hypothèse la
justification donnée à la ligne~$!A_k$, en citant, si nécessaire,
les prémisses de cette justification dans le premier bloc. Les
autres lignes gardent leur justification après renumérotation.
\iftag{FOL}{Les hypothèses de la suite réunie sont parmi celles des
deux blocs. Le choix des constantes propres préserve donc toutes
les applications de~\QR.}{} Nous avons ainsi une !!{derivation}
de~$!B$ à partir de~$\Gamma \cup \Delta$.
\end{proof}

En particulier, si $\Gamma \Proves !A$ et $!A \Proves !B$, alors
$\Gamma \Proves !B$. Il s'ensuit également que, si
$!A_1, \dots, !A_n \Proves !B$ et si $\Gamma \Proves !A_i$ pour
chaque~$i$, alors $\Gamma \Proves !B$.

\ollabel{prop:incons}
$\Gamma$ est incohérent si et seulement si
$\Gamma \Proves !A$ pour toute !!{formula}~$!A$.
\end{prop}

\begin{prop}[Compacité]
\ollabel{prop:proves-compact}
\begin{enumerate}
\item Si $\Gamma \Proves !A$, alors il existe une partie finie
  $\Gamma_0 \subseteq \Gamma$ telle que $\Gamma_0 \Proves !A$.
\item Si toute partie finie de~$\Gamma$ est cohérente, alors
  $\Gamma$ est cohérent.
\end{enumerate}
\end{prop}

\begin{proof}
\begin{enumerate}
\item Si $\Gamma \Proves !A$, soit $\Gamma_0$ l'ensemble fini des
  hypothèses d'une !!{derivation} de~$!A$ à partir de~$\Gamma$.
  Nous avons $\Gamma_0\subseteq\Gamma$. Les mêmes lignes, avec les
  mêmes justifications, donnent $\Gamma_0 \Proves !A$.
\item C'est la contraposée de~(1) dans le cas particulier où
  $!A \ident \lfalse$.
\end{enumerate}
\end{proof}
