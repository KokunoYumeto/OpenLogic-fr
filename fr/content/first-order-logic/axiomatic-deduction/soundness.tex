\olsection{Correction}

\begin{explain}
Un système de !!{derivation}, tel que la déduction axiomatique, est
\emph{correct} s'il ne permet pas de !!{derive} des résultats qui ne
valent pas réellement. La correction constitue ainsi une garantie de
fiabilité pour les systèmes de !!{derivation}. Selon la propriété de
théorie de la démonstration considérée, nous voulons notamment savoir
que :
\begin{enumerate}
\item toute !!{formula}~$!A$ !!{derivable} sans hypothèses est valide;
\item si $!A$ est !!{derivable} à partir d'autres !!{formula}s~$\Gamma$,
  elle en est également une conséquence;
\item si un ensemble de !!{formula}s~$\Gamma$ est incohérent, il est
  insatisfaisable.
\end{enumerate}
Ces propriétés sont essentielles pour un système de !!{derivation}.
Si l'une d'elles était en défaut, le système serait défectueux : il
permettrait de !!{derive} trop de résultats. Il est donc primordial
d'établir la correction d'un système de !!{derivation}.
\end{explain}

\begin{prop}
Si $!A$ est un axiome, alors
\iftag{FOL}
  {$\Sat{M}{!A}[s]$ pour toute !!{structure}~$\Struct{M}$ et toute
    assignation~$s$.}
  {$\pSat{v}{!A}$ pour toute !!{valuation}~$\pAssign{v}$.}
\end{prop}

\begin{proof}
\iftag{FOL}{Il faut vérifier que tous les axiomes sont valides.
  Traitons, par exemple, le cas de l'\olref[qua]{ax:q1}. Supposons que
  $t$ soit un terme clos et que
  $\Sat{M}{\lforall[x][!A]}[s]$. Par définition de la satisfaction,
  $\Sat{M}{!A}[s']$ pour toute $\varAssign{s'}{s}{x}$, et en
  particulier lorsque $s'(x) = \Value{t}{M}[s]$. D'après
  la \olref[syn][ext]{prop:ext-formulas},
  $\Sat{M}{\Subst{!A}{t}{x}}[s]$. Ainsi,
  $\Sat{M}{(\lforall[x][!A] \lif \Subst{!A}{t}{x})}[s]$.
  L'\olref[qua]{ax:q2} se vérifie de la même façon, et les
  axiomes propositionnels au moyen de leurs tables de vérité.}
  {Pour chaque axiome, sa table de vérité vérifie qu'il s'agit d'une
  tautologie.}
\end{proof}

\begin{thm}[Correction]
\ollabel{thm:soundness}
\iftag{FOL}{Soient $\Gamma$ un ensemble d'énoncés et $!A$ un énoncé.}{}
Si $\Gamma \Proves !A$, alors $\Gamma \Entails !A$.
\end{thm}

\begin{proof}
Soit $\Gamma_0$ l'ensemble des hypothèses effectivement utilisées dans
une !!{derivation} de~$!A$ à partir de~$\Gamma$.
\iftag{FOL}{Nous montrons, par récurrence sur la longueur de la
!!{derivation}, que pour toute !!{structure}~$\Struct{M}$ et toute
assignation~$s$ qui satisfont les hypothèses, la conclusion est
satisfaite sous~$s$. Au besoin, on prolonge l'interprétation aux
constantes intermédiaires. La vérité des hypothèses et de la conclusion
ne dépend que des symboles de leur langage.

Une ligne justifiée comme hypothèse est satisfaite par choix de
$\Struct{M}$ et de~$s$. Un axiome est satisfait d'après la proposition
précédente. Dans le cas d'un modus ponens, l'hypothèse de récurrence
s'applique aux préfixes qui établissent $!B$ et $!B \lif !A$. Ces
préfixes utilisent des hypothèses de~$\Gamma_0$. Comme
$\Sat{M}{!B}[s]$ et $\Sat{M}{!B \lif !A}[s]$, la définition de la
satisfaction de l'implication donne $\Sat{M}{!A}[s]$.

Supposons que la dernière ligne soit obtenue par la règle
universelle~\QR. La prémisse est $!C \lif !B(c)$ et la conclusion
$!C \lif \lforall[x][!B(x)]$. La constante~$c$ est absente de
$\Gamma_0$, de~$!C$ et de~$!B(x)$. Supposons $\Sat{M}{!C}[s]$ et
prenons une assignation $s'$ telle que $\varAssign{s'}{s}{x}$.
Soit $\Struct{M'}$ la structure qui ne diffère de~$\Struct{M}$
que par l'interprétation de~$c$, posée égale à~$s'(x)$.
Par la \olref[syn][ext]{prop:extensionality}, toutes les hypothèses
et $!C$ sont encore satisfaites dans~$\Struct{M'}$ sous l'assignation
initiale~$s$. Par l'hypothèse de récurrence appliquée à la prémisse,
$\Sat{M'}{!C \lif !B(c)}[s]$. On obtient donc $\Sat{M'}{!B(c)}[s]$.
Comme $!B(c)$ abrège $\Subst{!B(x)}{c}{x}$ et que
$\Value{c}{M'}[s]=s'(x)$, la \olref[syn][ext]{prop:ext-formulas}
donne $\Sat{M'}{!B(x)}[s']$. La \olref[syn][ext]{prop:extensionality}
donne alors $\Sat{M}{!B(x)}[s']$. L'assignation~$s'$ étant arbitraire,
la \olref[syn][ass]{prop:sat-quant} donne
$\Sat{M}{\lforall[x][!B(x)]}[s]$. Ainsi, l'implication cherchée est
satisfaite sous~$s$.

Nous laissons en exercice le cas de la règle existentielle~\QR{},
dont la conclusion est $\lexists[x][!B(x)] \lif !C$.
Quand les hypothèses et la conclusion sont des énoncés, la
\olref[syn][ass]{prop:sentence-sat-true} ramène ce résultat à la
satisfaction des énoncés. Toute structure qui satisfait $\Gamma$
satisfait aussi~$\Gamma_0$, donc $\Gamma \Entails !A$.}
{Procédons par récurrence sur la longueur de la !!{derivation}.
Si $!A$ est un axiome, c'est une tautologie d'après la proposition
précédente. Si $!A$ est une hypothèse, il est satisfait par toute
valuation qui satisfait~$\Gamma_0$.

Dans le cas du modus ponens, les deux préfixes qui établissent
$!B$ et $!B \lif !A$ sont plus courts et utilisent des hypothèses
de~$\Gamma_0$. Par récurrence, toute valuation qui satisfait
$\Gamma_0$ satisfait ces deux !!{formula}s. Elle satisfait
donc~$!A$ par la table de vérité de l'implication. Comme
$\Gamma_0\subseteq\Gamma$, on a $\Gamma \Entails !A$.}
\end{proof}

\tagprob{FOL}
\begin{prob}
Achever la preuve du \olref[fol][axd][sou]{thm:soundness}.
\end{prob}
\tagendprob

\begin{cor}
\ollabel{cor:weak-soundness}
Si $\Proves !A$, alors $!A$ est
\iftag{FOL}{satisfaite dans toute structure sous toute assignation}{une tautologie}.
\end{cor}

\begin{cor}
\ollabel{cor:consistency-soundness}
Si $\Gamma$ est satisfaisable, alors il est cohérent.
\end{cor}

\begin{proof}
Démontrons la contraposée. Supposons que $\Gamma$ ne soit pas
cohérent. Alors $\Gamma \Proves \lfalse$, c'est-à-dire qu'il existe
une !!{derivation} de~$\lfalse$ à partir de~$\Gamma$. D'après
le \olref{thm:soundness}, toute
\iftag{FOL}{!!{structure}~$\Struct{M}$}{!!{valuation}~$\pAssign{v}$}
qui satisfait~$\Gamma$ doit satisfaire~$\lfalse$. Or
\iftag{FOL}{$\Sat/{M}{\lfalse}$ pour toute
  !!{structure}~$\Struct{M}$}{$\pSat/{v}{\lfalse}$ pour toute
  !!{valuation}~$\pAssign{v}$}. Aucune
\iftag{FOL}{structure~$\Struct{M}$}{valuation~$\pAssign{v}$} ne peut
donc satisfaire~$\Gamma$; ainsi, $\Gamma$ n'est pas satisfaisable.
\end{proof}
