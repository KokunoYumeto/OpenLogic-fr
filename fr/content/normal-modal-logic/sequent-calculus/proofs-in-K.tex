\iftag{prvBox}{

\begin{ex}
  Voici une !!{derivation} en calcul des séquents qui montre que $\Proves (\Box!A \land \Box!B)
  \lif \Box (!A \land !B)$.
  \begin{prooftree}
    \Axiom$!A \fCenter !A$
    \doubleLine
    \UnaryInf$!B, !A \fCenter !A$   
    \Axiom$!B \fCenter !B$
    \doubleLine
    \UnaryInf$!B, !A \fCenter !B$   
    \RightLabel{\RightR{\land}}
    \BinaryInf$!B, !A \fCenter !A \land !B$    
    \RightLabel{$\Box$}
    \UnaryInf$\Box!B, \Box!A \fCenter \Box
    (!A \land !B)$
    \RightLabel{\LeftR{\land}}
    \UnaryInf$\Box!A \land \Box!B, \Box!A \fCenter \Box
    (!A \land !B)$
    \RightLabel{\LeftR{\Exchange}}
    \UnaryInf$\Box!A, \Box!A \land \Box!B \fCenter \Box
    (!A \land !B)$
    \RightLabel{\LeftR{\land}}
    \UnaryInf$\Box!A \land \Box!B, \Box!A \land \Box!B \fCenter \Box (!A \land !B)$
    \RightLabel{\LeftR{\Contraction}}
    \UnaryInf$\Box!A \land \Box!B \fCenter \Box (!A \land !B)$
    \RightLabel{\RightR{\lif}}
    \UnaryInf$\fCenter (\Box!A \land \Box!B)
    \lif \Box (!A \land !B)$
  \end{prooftree}
\end{ex}
}{}

\Writetofile{\jobname}{\string\Needspace{8\string\baselineskip}}
\begin{prob}
  Donnez des démonstrations en calcul des séquents dans~$\Log{K}$
  des !!{formula}s suivantes :
  \begin{enumerate}
    \item $\Box \lnot p \lif \Box(p \lif q)$
    \item $(\Box p \lor \Box q) \lif \Box(p \lor q)$
    \item $\Diamond p \lif \Diamond(p \lor q)$
    \item $\Box(p \land q) \lif \Box p$
  \end{enumerate}
\end{prob}
\begin{editorial}Dans la démonstration de la dualité, les deux passages d’une formule à droite à sa négation à gauche sont étiquetés comme règles de négation à gauche, en corrigeant les deux étiquettes source. Dans cette démonstration, le biconditionnel final est traité comme l’abréviation de la conjonction des deux implications ; cette lecture justifie la règle de conjonction indiquée.\end{editorial}
