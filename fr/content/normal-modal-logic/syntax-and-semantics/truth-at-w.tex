\olsection{Vérité en un monde}

Tout modèle modal détermine quelles !!{formula}s modales sont vraies
en chacun de ses mondes. La relation « le modèle~$\mModel{M}$ rend
la !!{formula}~$!A$ vraie au monde~$w$ » est la notion fondamentale
de la sémantique relationnelle. Elle se définit par induction et,
pour les connecteurs non modaux, coïncide avec les clauses usuelles
des tables de vérité.

\begin{defn}\ollabel{defn:mmodels}
  La \emph{vérité d'!!a{formula}~$!A$ en~$w$} dans un modèle
  $\mModel M$, notée $\mSat{M}{!A}[w]$, se définit par induction :
  \begin{enumerate}
  \tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\mSat{M}{\lfalse}[w]$ ne vaut jamais}.}{}
  \tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\mSat{M}{\ltrue}[w]$ vaut toujours}.}{}
  \item $\mSat{M}{p}[w]$ si et seulement si $w \in V(p)$.
  \tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ si et seulement si
    $\mSat/{M}{!B}[w]$}.}{}
  \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\mSat{M}{\indfrm}[w]$ si et seulement si
    $\mSat{M}{!B}[w]$ et $\mSat{M}{!C}[w]$}.}{}
  \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\mSat{M}{\indfrm}[w]$ si et seulement si
    $\mSat{M}{!B}[w]$ ou $\mSat{M}{!C}[w]$} (ou les deux).}{}
  \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\mSat{M}{\indfrm}[w]$ si et seulement si
    $\mSat/{M}{!B}[w]$ ou $\mSat{M}{!C}[w]$}.}{}
  \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\mSat{M}{\indfrm}[w]$ si et seulement si
    soit $\mSat{M}{!B}[w]$ et $\mSat{M}{!C}[w]$,
    soit ni $\mSat{M}{!B}[w]$ ni $\mSat{M}{!C}[w]$}.}{}
  \item\ollabel{defn:sub:mmodels-box}
    \indcase{!A}{\Box !B}{$\mSat{M}{\indfrm}[w]$ si et seulement si
    $\mSat{M}{!B}[w']$ pour tout $w' \in W$ tel que $Rww'$.}
  \item\ollabel{defn:sub:mmodels-diamond}
    \indcase{!A}{\Diamond !B}{$\mSat{M}{\indfrm}[w]$ si et seulement si
    $\mSat{M}{!B}[w']$ pour au moins un $w' \in W$ tel que $Rww'$.}
  \end{enumerate}
\end{defn}

Selon la clause~\olref{defn:sub:mmodels-box}, !!a{formula}~$\Box
!B$ est vraie en~$w$ lorsqu'aucun monde~$w'$ ne vérifie $Rww'$.
Dans ce cas, $\Box !B$ est vraie \emph{par vacuité} en~$w$.
Il se peut aussi que $\Box !B$ soit vraie en~$w$ alors que
$!B$ ne l'est pas. De même, la vérité de~$!B$ en~$w$
ne garantit pas celle de~$\Diamond !B$ en~$w$.
Cette dernière implication vaut toutefois si $Rww$, par exemple
si $R$ est réflexive. S'il n'existe aucun~$w'$ tel que $Rww'$,
alors $\mSat/{M}{\Diamond !A}[w]$ pour toute formule~$!A$.

\begin{prob}
  Considérons le modèle de la~\olref[nml][syn][rel]{fig:simple}.
  Lesquelles des assertions suivantes sont vraies ?
    \begin{enumerate}
    \item $\mSat{M}{q}[w_1]$;
    \item $\mSat{M}{\lnot q}[w_3]$;
    \item $\mSat{M}{p \lor q}[w_1]$;
    \item $\mSat{M}{\Box (p \lor q)}[w_1]$;
    \item $\mSat{M}{\Box q}[w_3]$;
    \item $\mSat{M}{\Box \bot}[w_3]$;
    \item $\mSat{M}{\Diamond q}[w_1]$;
    \item $\mSat{M}{\Box q}[w_1]$;
    \item $\mSat{M}{\lnot \Box\Box \lnot q}[w_1]$.
    \end{enumerate}
\end{prob}

% Prove duality of \Box and \Diamond
\begin{tagblock}{prvDiamond}
\begin{prop}\ollabel{prop:dual}
  \begin{enumerate}
  \item $\mSat{M}{\Box !A}[w]$ si et seulement si
    $\mSat{M}{\lnot\Diamond\lnot !A}[w]$.
  \item $\mSat{M}{\Diamond !A}[w]$ si et seulement si
    $\mSat{M}{\lnot\Box\lnot !A}[w]$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ si et seulement si
      $\mSat/{M}{\Diamond\lnot !A}[w]$, par définition de
      $\mSat{M}{}[w]$. La relation
      $\mSat{M}{\Diamond\lnot !A}[w]$ vaut si et seulement s'il
      existe un~$w'$ tel que $Rww'$ et
      $\mSat{M}{\lnot !A}[w']$. Donc
      $\mSat/{M}{\Diamond\lnot !A}[w]$ si et seulement si, pour
      tout~$w'$ tel que $Rww'$, on a
      $\mSat/{M}{\lnot !A}[w']$. Or
      $\mSat/{M}{\lnot !A}[w']$ si et seulement si
      $\mSat{M}{!A}[w']$. Ainsi,
      $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ si et seulement si
      $\mSat{M}{!A}[w']$ pour tout~$w'$ tel que $Rww'$.
      D'après la définition de $\mSat{M}{}[w]$, cela équivaut à
      $\mSat{M}{\Box !A}[w]$.
    \item \iftag{probDiamond}{Exercice.}{$\mSat{M}{\lnot\Box\lnot !A}[w]$
      si et seulement si $\mSat/{M}{\Box\lnot !A}[w]$.
      La relation $\mSat{M}{\Box\lnot !A}[w]$ vaut si et
      seulement si $\mSat{M}{\lnot !A}[w']$ pour tout~$w'$
      tel que $Rww'$. Donc
      $\mSat/{M}{\Box\lnot !A}[w]$ si et seulement s'il existe
      un~$w'$ tel que $Rww'$ et
      $\mSat/{M}{\lnot !A}[w']$. Or
      $\mSat/{M}{\lnot !A}[w']$ si et seulement si
      $\mSat{M}{!A}[w']$. Ainsi,
      $\mSat{M}{\lnot\Box\lnot !A}[w]$ si et seulement s'il existe
      un~$w'$ tel que $Rww'$ et $\mSat{M}{!A}[w']$.
      Par définition de $\mSat{M}{}[w]$, cela équivaut à
      $\mSat{M}{\Diamond !A}[w]$.}
  \end{enumerate}
\end{proof}

\begin{probtag}{probDiamond}
  Complétez la démonstration de la~\olref[nml][syn][trw]{prop:dual}.
\end{probtag}
\end{tagblock}

\begin{prob}
  Soit $\mModel{M} = \tuple{W, R, V}$ un modèle. Supposons que
  $w_1, w_2 \in W$ vérifient les conditions suivantes :
  \begin{enumerate}
  \item $w_1 \in V(p)$ si et seulement si $w_2 \in V(p)$, pour
    toute !!{propositional variable}~$p$;
  \item pour tout $w \in W$, $Rw_1w$ si et seulement si $Rw_2w$.
  \end{enumerate}
  Par induction sur les !!{formula}s, montrez que, pour toute
  !!{formula}~$!A$, $\mSat{M}{!A}[w_1]$ si et seulement si
  $\mSat{M}{!A}[w_2]$.
\end{prob}

\begin{probtag}{prvDiamond}
  Soit $\mModel{M} = \tuple{W, R, V}$. Montrez que
  $\mSat{M}{\lnot\Diamond !A}[w]$ si et seulement si
  $\mSat{M}{\Box\lnot!A}[w]$.
\end{probtag}
